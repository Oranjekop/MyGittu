cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(MyGittuShell LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(MyGittuShell SHARED shell.cpp shell.def shell.rc)
target_compile_features(MyGittuShell PRIVATE cxx_std_17)
target_compile_definitions(MyGittuShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(MyGittuShell PRIVATE shell32 ole32 advapi32 user32 uuid)
add_executable(MyGittuShellSmoke smoke.cpp)
target_compile_features(MyGittuShellSmoke PRIVATE cxx_std_17)
target_link_libraries(MyGittuShellSmoke PRIVATE shell32 ole32 uuid)
if(MSVC)
 target_compile_options(MyGittuShell PRIVATE /W4 /permissive- /analyze /utf-8)
 target_compile_options(MyGittuShellSmoke PRIVATE /W4 /utf-8)
endif()
